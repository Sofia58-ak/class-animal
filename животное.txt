# формат отношений "a знает о b, b знает о a, a знает о c, но c не знает о b".
# цепочка наследования из 3 классов, наш класс конечный, но как его могут 
# дополнять предыдущие так и он может дополнять предыдущие.
# базовый класс - млекопитающее - Mammal.
# protected - информация которую можно изменить изнутри
# public - информация которую можно изменить из вне.
---------------------------------------------------------------

base class Mammal  #Млекопитающее
{
	protected:  #свойства базовые
	
	string Name;              #имя
	float  BodyTemperature;   #температура тела

	public:  #методы базовые
	
	nurse_young();    #кормить детёнышей молоком
	breathe();        #дышать воздухом
	}

class AquaticMammal  #Водное млекопитающее
{
	protected:  #свойства базовые
	
	string Habitat;    #среда обитания (океан, море, река)
	float  MaxDepth;   #максимальная глубина погружения
	
	protected:  #свойства унаследованные
        
	string Name;              #имя                от Mammal
	float  BodyTemperature;   #температура тела    от Mammal
	
	public:  #методы
        
	#свои - новые
	void dive();              #нырять на глубину
	
	#унаследованные:
	nurse_young();            #кормить детёнышей   от Mammal
	breathe();                #дышать воздухом     от Mammal
	}

class Dolphin  #Дельфин (наш конечный класс)
{
	protected:  #свойства базовые
	
	#свои - новые
	ushort PodSize;           #размер стаи (количество особей)

	#унаследованные:
	string Name;              #имя                от Mammal
	float  BodyTemperature;   #температура тела    от Mammal
	string Habitat;           #среда обитания      от AquaticMammal
	float  MaxDepth;          #максимальная глубина от AquaticMammal

	public: #методы
	
	#свои - новые
	void echolocate();        #эхолокация (звуковые щелчки)
        void communicate();       #общение со стаей (свисты)

	#изменённые:
        void breathe() override;  #дышать через дыхало на поверхности
        
	#унаследованные:
        void nurse_young();       #кормить детёнышей        от Mammal
        void dive();              #нырять                   от AquaticMammal
	}

---------------------------------------------------------------
# ПОЯСНЕНИЕ ОТНОШЕНИЙ "ЗНАЕТ О":
#
# Mammal знает о себе (Name, BodyTemperature, nurse_young, breathe)
# AquaticMammal знает о Mammal (наследует его свойства и методы)
# Dolphin знает о AquaticMammal и Mammal (наследует всё от обоих)
#
# НО: Mammal НЕ знает о AquaticMammal и Dolphin
#     AquaticMammal НЕ знает о Dolphin
#
# Это односторонняя иерархия: потомки знают о предках, 
# но предки не знают о потомках.
---------------------------------------------------------------
